cmake_minimum_required(VERSION 3.10)
project(SimpleMap VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.9 REQUIRED COMPONENTS Widgets Svg)
add_executable(simplemap src/main.cpp)
target_link_libraries(simplemap PRIVATE Qt5::Widgets Qt5::Svg)
if(WIN32)
  set_target_properties(simplemap PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS simplemap RUNTIME DESTINATION bin)
