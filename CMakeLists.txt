cmake_minimum_required(VERSION 3.10)
project(SimpleMap VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.9 REQUIRED COMPONENTS Widgets Svg)

qt5_add_resources(zhili_template_resources assets/templates.qrc)
add_library(zhili_core STATIC src/document.cpp src/layout.cpp src/clipboard.cpp
  src/pptx.cpp src/importer.cpp src/templates.cpp ${zhili_template_resources})
target_link_libraries(zhili_core PUBLIC Qt5::Core)
qt5_add_resources(zhili_font_resources assets/fonts.qrc)
add_library(zhili_appearance STATIC src/appearance.cpp ${zhili_font_resources})
target_link_libraries(zhili_appearance PUBLIC Qt5::Widgets)
qt5_add_resources(zhili_symbol_resources assets/symbols.qrc)
add_library(zhili_symbols STATIC src/symbols.cpp src/symbolpicker.cpp ${zhili_symbol_resources})
target_link_libraries(zhili_symbols PUBLIC Qt5::Widgets Qt5::Svg)
add_library(zhili_templates_ui STATIC src/template_dialog.cpp)
target_link_libraries(zhili_templates_ui PUBLIC zhili_core Qt5::Widgets)
add_executable(simplemap WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(simplemap PRIVATE zhili_core zhili_appearance zhili_symbols zhili_templates_ui Qt5::Svg)
set_target_properties(simplemap PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER org.zhili.simplemap
  MACOSX_BUNDLE_BUNDLE_NAME ZhiLi
  MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")

if(MSVC)
  target_compile_options(zhili_core PRIVATE /utf-8)
  target_compile_options(zhili_appearance PRIVATE /utf-8)
  target_compile_options(zhili_symbols PRIVATE /utf-8)
  target_compile_options(zhili_templates_ui PRIVATE /utf-8)
  target_compile_options(simplemap PRIVATE /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Qt5 5.9 REQUIRED COMPONENTS Test)
  add_executable(zhili_smoke src/main.cpp)
  target_compile_definitions(zhili_smoke PRIVATE ZHILI_TEST)
  target_link_libraries(zhili_smoke PRIVATE zhili_core zhili_appearance zhili_symbols zhili_templates_ui Qt5::Svg Qt5::Test)
  add_executable(zhili_document_test tests/document_test.cpp)
  target_link_libraries(zhili_document_test PRIVATE zhili_core)
  add_executable(zhili_layout_test tests/layout_test.cpp)
  target_link_libraries(zhili_layout_test PRIVATE zhili_core)
  add_executable(zhili_clipboard_test tests/clipboard_test.cpp)
  target_link_libraries(zhili_clipboard_test PRIVATE zhili_core)
  add_executable(zhili_symbol_document_test tests/symbol_document_test.cpp)
  target_link_libraries(zhili_symbol_document_test PRIVATE zhili_core)
  add_executable(zhili_pptx_test tests/pptx_test.cpp)
  target_link_libraries(zhili_pptx_test PRIVATE zhili_core)
  add_executable(zhili_importer_test tests/importer_test.cpp)
  target_link_libraries(zhili_importer_test PRIVATE zhili_core)
  add_executable(zhili_templates_test tests/templates_test.cpp)
  target_link_libraries(zhili_templates_test PRIVATE zhili_core)
  add_executable(zhili_template_dialog_test tests/template_dialog_test.cpp)
  target_link_libraries(zhili_template_dialog_test PRIVATE zhili_templates_ui zhili_appearance Qt5::Test)
  if(MSVC)
    target_compile_options(zhili_smoke PRIVATE /utf-8)
    target_compile_options(zhili_document_test PRIVATE /utf-8)
    target_compile_options(zhili_layout_test PRIVATE /utf-8)
    target_compile_options(zhili_clipboard_test PRIVATE /utf-8)
    target_compile_options(zhili_symbol_document_test PRIVATE /utf-8)
    target_compile_options(zhili_pptx_test PRIVATE /utf-8)
    target_compile_options(zhili_importer_test PRIVATE /utf-8)
    target_compile_options(zhili_templates_test PRIVATE /utf-8)
    target_compile_options(zhili_template_dialog_test PRIVATE /utf-8)
  endif()
  add_executable(zhili_symbols_test tests/symbols_test.cpp)
  target_link_libraries(zhili_symbols_test PRIVATE zhili_symbols Qt5::Test)
  if(MSVC)
    target_compile_options(zhili_symbols_test PRIVATE /utf-8)
  endif()
  add_test(NAME zhili_symbols COMMAND zhili_symbols_test)
  add_test(NAME zhili_document COMMAND zhili_document_test)
  add_test(NAME zhili_layout COMMAND zhili_layout_test)
  add_test(NAME zhili_clipboard COMMAND zhili_clipboard_test)
  add_test(NAME zhili_symbol_document COMMAND zhili_symbol_document_test)
  add_test(NAME zhili_pptx COMMAND zhili_pptx_test)
  add_test(NAME zhili_importer COMMAND zhili_importer_test)
  add_test(NAME zhili_templates COMMAND zhili_templates_test)
  add_test(NAME zhili_template_dialog COMMAND zhili_template_dialog_test)
  add_test(NAME zhili_template_dialog_large_type COMMAND zhili_template_dialog_test)
  set(zhili_test_platform offscreen)
  if(WIN32)
    # Match the portable build's Windows QPA and bundled CJK font engine.
    set(zhili_test_platform "windows:fontengine=freetype")
  endif()
  set_tests_properties(zhili_symbols zhili_template_dialog PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=${zhili_test_platform};QT_FORCE_STDERR_LOGGING=1;QT_FONT_DPI=96" TIMEOUT 30)
  set_tests_properties(zhili_template_dialog_large_type PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=${zhili_test_platform};QT_FONT_DPI=96;QT_SCALE_FACTOR=2;ZHILI_TEMPLATE_FONT_POINTS=22" TIMEOUT 30)
  foreach(scale IN ITEMS 1 1.25 1.5 2)
    add_test(NAME zhili_smoke_${scale}x COMMAND zhili_smoke)
    set_tests_properties(zhili_smoke_${scale}x PROPERTIES
      ENVIRONMENT "QT_QPA_PLATFORM=${zhili_test_platform};QT_FORCE_STDERR_LOGGING=1;QT_FONT_DPI=96;QT_SCALE_FACTOR=${scale};ZHILI_EXPECT_SCALE=${scale}" TIMEOUT 30)
  endforeach()
endif()

install(TARGETS simplemap RUNTIME DESTINATION bin BUNDLE DESTINATION .)
install(FILES assets/fonts/OFL-SourceHanSans.txt assets/fonts/OFL-SourceHanSerif.txt
  DESTINATION share/licenses/zhili)

install(FILES assets/symbols/LICENSE.txt DESTINATION share/licenses/zhili RENAME SYMBOLS-CC0.txt)
install(FILES src/vendor/miniz/LICENSE DESTINATION share/licenses/zhili RENAME MINIZ-MIT.txt)
